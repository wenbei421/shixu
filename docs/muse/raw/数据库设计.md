下面是一套 **可直接运行的 SQLite 原型数据库**，包含建表、种子数据和常用查询。目标：**能落地、能跑、能与前端原型字段对上**。

---

## 1. 快速开始

```bash
# 方式一：sqlite3 CLI
sqlite3 muse.db < muse_schema.sql
sqlite3 muse.db < muse_seed.sql

# 方式二：Node.js（better-sqlite3）
npm i better-sqlite3
node demo.js

# 方式三：DB Browser for SQLite 图形工具
# 新建 muse.db → 执行 SQL 文件
```

目录建议：

```
muse-prototype/
  ├─ muse_schema.sql     # 建表 + 视图 + 触发器
  ├─ muse_seed.sql       # 种子数据
  ├─ muse_queries.sql    # 常用查询示例
  └─ muse.db             # 生成的数据库
```

## 2. 精简版 Schema（`muse_schema.sql`）

> 原型只保留实际用到的表：**灵感、标签、关联、项目、回顾、设置**。  
> AI、向量、同步、历史版本等留到后续迭代。

```sql
-- =========================================================
-- 0. PRAGMA
-- =========================================================
PRAGMA foreign_keys = ON;
PRAGMA journal_mode = WAL;
PRAGMA synchronous = NORMAL;
PRAGMA encoding = 'UTF-8';

-- =========================================================
-- 1. 项目表
-- =========================================================
CREATE TABLE IF NOT EXISTS muse_projects (
  id          INTEGER PRIMARY KEY,              -- 雪花 ID
  name        TEXT    NOT NULL UNIQUE,
  description TEXT,
  color       TEXT,
  status      TEXT    NOT NULL DEFAULT 'active'
              CHECK (status IN ('active','paused','done','archived')),
  created_at  INTEGER NOT NULL,
  updated_at  INTEGER NOT NULL,
  deleted_at  INTEGER
);
CREATE INDEX IF NOT EXISTS idx_muse_projects_status  ON muse_projects(status);
CREATE INDEX IF NOT EXISTS idx_muse_projects_deleted ON muse_projects(deleted_at);

-- =========================================================
-- 2. 灵感主表
-- =========================================================
CREATE TABLE IF NOT EXISTS muse_notes (
  id          INTEGER PRIMARY KEY,              -- 雪花 ID
  content     TEXT    NOT NULL,
  status      TEXT    NOT NULL DEFAULT 'new'
              CHECK (status IN ('new','check','doing','done')),
  priority    INTEGER NOT NULL DEFAULT 0 CHECK (priority BETWEEN 0 AND 3),
  source      TEXT    NOT NULL DEFAULT 'manual'
              CHECK (source IN ('quick','manual','clip','book','pod','web','api','import')),
  source_url  TEXT,
  project_id  INTEGER REFERENCES muse_projects(id) ON DELETE SET NULL,
  pinned      INTEGER NOT NULL DEFAULT 0 CHECK (pinned IN (0,1)),
  archived_at INTEGER,
  deleted_at  INTEGER,
  created_at  INTEGER NOT NULL,
  updated_at  INTEGER NOT NULL
);
CREATE INDEX IF NOT EXISTS idx_muse_notes_status      ON muse_notes(status);
CREATE INDEX IF NOT EXISTS idx_muse_notes_project_id  ON muse_notes(project_id);
CREATE INDEX IF NOT EXISTS idx_muse_notes_created_at  ON muse_notes(created_at DESC);
CREATE INDEX IF NOT EXISTS idx_muse_notes_updated_at  ON muse_notes(updated_at DESC);
CREATE INDEX IF NOT EXISTS idx_muse_notes_archived_at ON muse_notes(archived_at);
CREATE INDEX IF NOT EXISTS idx_muse_notes_deleted_at  ON muse_notes(deleted_at);

-- =========================================================
-- 3. 标签表
-- =========================================================
CREATE TABLE IF NOT EXISTS muse_tags (
  id          INTEGER PRIMARY KEY,              -- 雪花 ID
  name        TEXT    NOT NULL UNIQUE,
  color       TEXT,
  parent_id   INTEGER REFERENCES muse_tags(id) ON DELETE SET NULL,
  created_at  INTEGER NOT NULL,
  updated_at  INTEGER NOT NULL
);
CREATE INDEX IF NOT EXISTS idx_muse_tags_parent_id ON muse_tags(parent_id);

-- =========================================================
-- 4. 灵感-标签 多对多
-- =========================================================
CREATE TABLE IF NOT EXISTS muse_note_tags (
  note_id     INTEGER NOT NULL REFERENCES muse_notes(id) ON DELETE CASCADE,
  tag_id      INTEGER NOT NULL REFERENCES muse_tags(id)  ON DELETE CASCADE,
  created_at  INTEGER NOT NULL,
  PRIMARY KEY (note_id, tag_id)
);
CREATE INDEX IF NOT EXISTS idx_muse_note_tags_tag_id ON muse_note_tags(tag_id);

-- =========================================================
-- 5. 回顾记录
-- =========================================================
CREATE TABLE IF NOT EXISTS muse_reviews (
  id              INTEGER PRIMARY KEY,          -- 雪花 ID
  note_id         INTEGER NOT NULL REFERENCES muse_notes(id) ON DELETE CASCADE,
  reviewed_at     INTEGER NOT NULL,
  result          TEXT NOT NULL
                  CHECK (result IN ('keep','archive','promote','merge','delete','snooze')),
  next_review_at  INTEGER,
  note            TEXT,
  created_at      INTEGER NOT NULL
);
CREATE INDEX IF NOT EXISTS idx_muse_reviews_note_id        ON muse_reviews(note_id);
CREATE INDEX IF NOT EXISTS idx_muse_reviews_next_review_at ON muse_reviews(next_review_at);

-- =========================================================
-- 6. 设置表
-- =========================================================
CREATE TABLE IF NOT EXISTS muse_settings (
  key         TEXT PRIMARY KEY,
  value       TEXT,
  updated_at  INTEGER NOT NULL
);

-- =========================================================
-- 7. FTS5 全文搜索
-- =========================================================
CREATE VIRTUAL TABLE IF NOT EXISTS muse_notes_fts USING fts5(
  note_id UNINDEXED,
  content,
  tags_text,
  project_name,
  tokenize = 'unicode61 remove_diacritics 2'
);

-- =========================================================
-- 8. 触发器：muse_notes -> FTS
-- =========================================================
CREATE TRIGGER IF NOT EXISTS trg_muse_notes_ai AFTER INSERT ON muse_notes BEGIN
  INSERT INTO muse_notes_fts(note_id, content, tags_text, project_name)
  VALUES (
    NEW.id,
    NEW.content,
    '',
    COALESCE((SELECT name FROM muse_projects WHERE id = NEW.project_id), '')
  );
END;

CREATE TRIGGER IF NOT EXISTS trg_muse_notes_au AFTER UPDATE ON muse_notes BEGIN
  DELETE FROM muse_notes_fts WHERE note_id = NEW.id;
  INSERT INTO muse_notes_fts(note_id, content, tags_text, project_name)
  SELECT
    NEW.id,
    NEW.content,
    COALESCE((
      SELECT group_concat(t.name, ' ')
      FROM muse_note_tags nt
      JOIN muse_tags t ON t.id = nt.tag_id
      WHERE nt.note_id = NEW.id
    ), ''),
    COALESCE((SELECT name FROM muse_projects WHERE id = NEW.project_id), '');
END;

CREATE TRIGGER IF NOT EXISTS trg_muse_notes_ad AFTER DELETE ON muse_notes BEGIN
  DELETE FROM muse_notes_fts WHERE note_id = OLD.id;
END;

-- =========================================================
-- 9. 触发器：muse_note_tags -> FTS
-- =========================================================
CREATE TRIGGER IF NOT EXISTS trg_muse_note_tags_ai AFTER INSERT ON muse_note_tags BEGIN
  DELETE FROM muse_notes_fts WHERE note_id = NEW.note_id;
  INSERT INTO muse_notes_fts(note_id, content, tags_text, project_name)
  SELECT
    n.id,
    n.content,
    COALESCE((
      SELECT group_concat(t.name, ' ')
      FROM muse_note_tags nt
      JOIN muse_tags t ON t.id = nt.tag_id
      WHERE nt.note_id = n.id
    ), ''),
    COALESCE((SELECT name FROM muse_projects WHERE id = n.project_id), '')
  FROM muse_notes n
  WHERE n.id = NEW.note_id;
END;

CREATE TRIGGER IF NOT EXISTS trg_muse_note_tags_ad AFTER DELETE ON muse_note_tags BEGIN
  DELETE FROM muse_notes_fts WHERE note_id = OLD.note_id;
  INSERT INTO muse_notes_fts(note_id, content, tags_text, project_name)
  SELECT
    n.id,
    n.content,
    COALESCE((
      SELECT group_concat(t.name, ' ')
      FROM muse_note_tags nt
      JOIN muse_tags t ON t.id = nt.tag_id
      WHERE nt.note_id = n.id
    ), ''),
    COALESCE((SELECT name FROM muse_projects WHERE id = n.project_id), '')
  FROM muse_notes n
  WHERE n.id = OLD.note_id;
END;

-- =========================================================
-- 10. 视图（v_muse_ 前缀）
-- =========================================================
DROP VIEW IF EXISTS v_muse_notes_full;
CREATE VIEW v_muse_notes_full AS
SELECT
  n.*,
  p.name AS project_name,
  COALESCE((
    SELECT group_concat(t.name, ',')
    FROM muse_note_tags nt
    JOIN muse_tags t ON t.id = nt.tag_id
    WHERE nt.note_id = n.id
  ), '') AS tags_text
FROM muse_notes n
LEFT JOIN muse_projects p ON p.id = n.project_id
WHERE n.deleted_at IS NULL;

DROP VIEW IF EXISTS v_muse_inbox;
CREATE VIEW v_muse_inbox AS
SELECT * FROM v_muse_notes_full
WHERE archived_at IS NULL AND status = 'new';

DROP VIEW IF EXISTS v_muse_today;
CREATE VIEW v_muse_today AS
SELECT * FROM v_muse_notes_full
WHERE archived_at IS NULL
  AND created_at >= (CAST(strftime('%s','now','start of day') AS INTEGER) * 1000);

DROP VIEW IF EXISTS v_muse_unsorted;
CREATE VIEW v_muse_unsorted AS
SELECT * FROM v_muse_notes_full
WHERE archived_at IS NULL
  AND (tags_text IS NULL OR tags_text = '')
  AND project_id IS NULL;

DROP VIEW IF EXISTS v_muse_archive;
CREATE VIEW v_muse_archive AS
SELECT * FROM v_muse_notes_full
WHERE archived_at IS NOT NULL;

DROP VIEW IF EXISTS v_muse_review_due;
CREATE VIEW v_muse_review_due AS
SELECT n.*, r.next_review_at, r.result
FROM muse_notes n
JOIN muse_reviews r ON r.note_id = n.id
WHERE n.deleted_at IS NULL
  AND n.archived_at IS NULL
  AND r.next_review_at IS NOT NULL
  AND r.next_review_at <= (CAST(strftime('%s','now') AS INTEGER) * 1000)
ORDER BY r.next_review_at ASC;

DROP VIEW IF EXISTS v_muse_tag_stats;
CREATE VIEW v_muse_tag_stats AS
SELECT
  t.id,
  t.name,
  t.color,
  COUNT(nt.note_id) AS note_count
FROM muse_tags t
LEFT JOIN muse_note_tags nt ON nt.tag_id = t.id
LEFT JOIN muse_notes n ON n.id = nt.note_id
  AND n.deleted_at IS NULL AND n.archived_at IS NULL
GROUP BY t.id;

DROP VIEW IF EXISTS v_muse_project_stats;
CREATE VIEW v_muse_project_stats AS
SELECT
  p.id,
  p.name,
  p.status,
  COUNT(n.id) AS note_count,
  SUM(CASE WHEN n.status = 'done' THEN 1 ELSE 0 END) AS done_count
FROM muse_projects p
LEFT JOIN muse_notes n ON n.project_id = p.id AND n.deleted_at IS NULL
GROUP BY p.id;
```

## 4. 常用查询（`muse_queries.sql`）

```sql
-- =========================================================
-- 1. 收件箱
-- =========================================================
SELECT id, content, status, project_name, tags_text, created_at
FROM v_muse_inbox
ORDER BY created_at DESC;

-- =========================================================
-- 2. 今日灵感
-- =========================================================
SELECT id, content, status, tags_text
FROM v_muse_today
ORDER BY created_at DESC;

-- =========================================================
-- 3. 待整理（无标签、无项目）
-- =========================================================
SELECT id, content, created_at
FROM v_muse_unsorted
ORDER BY created_at DESC;

-- =========================================================
-- 4. 归档
-- =========================================================
SELECT id, content, archived_at
FROM v_muse_archive
ORDER BY archived_at DESC;

-- =========================================================
-- 5. 按标签筛选
-- =========================================================
SELECT n.id, n.content, n.status
FROM muse_notes n
JOIN muse_note_tags nt ON nt.note_id = n.id
JOIN muse_tags t ON t.id = nt.tag_id
WHERE t.name = '产品'
  AND n.deleted_at IS NULL
  AND n.archived_at IS NULL
ORDER BY n.created_at DESC;

-- =========================================================
-- 6. 按项目筛选
-- =========================================================
SELECT n.id, n.content, n.status
FROM muse_notes n
JOIN muse_projects p ON p.id = n.project_id
WHERE p.name = '灵感工具 App'
  AND n.deleted_at IS NULL
ORDER BY n.created_at DESC;

-- =========================================================
-- 7. 全文搜索
-- =========================================================
SELECT n.id, n.content, n.status,
       snippet(muse_notes_fts, 1, '<mark>', '</mark>', '…', 12) AS snippet
FROM muse_notes_fts f
JOIN muse_notes n ON n.id = f.note_id
WHERE f MATCH '灵感'
  AND n.deleted_at IS NULL
  AND n.archived_at IS NULL
ORDER BY f.rank
LIMIT 20;

-- =========================================================
-- 8. 标签统计
-- =========================================================
SELECT name, note_count
FROM v_muse_tag_stats
ORDER BY note_count DESC, name;

-- =========================================================
-- 9. 项目统计
-- =========================================================
SELECT name, note_count, done_count
FROM v_muse_project_stats
ORDER BY note_count DESC;

-- =========================================================
-- 10. 看板视图（按状态分组）
-- =========================================================
SELECT status, COUNT(*) AS cnt
FROM muse_notes
WHERE deleted_at IS NULL AND archived_at IS NULL
GROUP BY status;

-- =========================================================
-- 11. 到期回顾
-- =========================================================
SELECT id, content, next_review_at, result
FROM v_muse_review_due;

-- =========================================================
-- 12. 插入灵感 + 绑定标签（事务示例）
-- =========================================================
BEGIN;

INSERT INTO muse_notes(id, content, status, source, created_at, updated_at)
VALUES (
  2346000000000000099,
  '这是一条新的灵感。',
  'new', 'quick',
  CAST(strftime('%s','now') AS INTEGER) * 1000,
  CAST(strftime('%s','now') AS INTEGER) * 1000
);

INSERT INTO muse_note_tags(note_id, tag_id, created_at)
SELECT 2346000000000000099, id, CAST(strftime('%s','now') AS INTEGER) * 1000
FROM muse_tags
WHERE name IN ('产品', '交互');

COMMIT;

-- =========================================================
-- 13. 更新状态
-- =========================================================
UPDATE muse_notes
SET status = 'done',
    updated_at = CAST(strftime('%s','now') AS INTEGER) * 1000
WHERE id = 2346000000000000001;

-- =========================================================
-- 14. 归档
-- =========================================================
UPDATE muse_notes
SET archived_at = CAST(strftime('%s','now') AS INTEGER) * 1000
WHERE id = 2346000000000000001;

-- =========================================================
-- 15. 软删除
-- =========================================================
UPDATE muse_notes
SET deleted_at = CAST(strftime('%s','now') AS INTEGER) * 1000
WHERE id = 2346000000000000001;
```

## 5. Node.js 调用示例（`demo.js`）

```js
const Database = require('better-sqlite3');
const db = new Database('muse.db');

// ---------- 常用查询 ----------
const getInbox = db.prepare(`
  SELECT id, content, status, project_name, tags_text, created_at
  FROM v_muse_inbox
  ORDER BY created_at DESC
`);

const searchNotes = db.prepare(`
  SELECT n.id, n.content, n.status
  FROM muse_notes_fts f
  JOIN muse_notes n ON n.id = f.note_id
  WHERE f MATCH ?
    AND n.deleted_at IS NULL
    AND n.archived_at IS NULL
  ORDER BY f.rank
  LIMIT 20
`);

const getTags = db.prepare(`
  SELECT name, note_count FROM v_muse_tag_stats ORDER BY note_count DESC
`);

// ---------- 新增灵感（雪花 ID 示例） ----------
function nextSnowflake() {
  // 简化版，仅演示：时间戳 + 固定 worker
  const EPOCH = 1704067200000n;
  const now = BigInt(Date.now());
  return (now - EPOCH) << 22n | (1n << 17n) | (1n << 12n);
}

const insertNote = db.prepare(`
  INSERT INTO muse_notes(id, content, status, source, created_at, updated_at)
  VALUES (?, ?, 'new', 'quick', ?, ?)
`);

const addTag = db.prepare(`
  INSERT OR IGNORE INTO muse_note_tags(note_id, tag_id, created_at)
  SELECT ?, id, ? FROM muse_tags WHERE name = ?
`);

function createNote(content, tags = []) {
  const id = Number(nextSnowflake());
  const now = Date.now();

  const tx = db.transaction(() => {
    insertNote.run(id, content, now, now);
    for (const t of tags) addTag.run(id, now, t);
  });

  tx();
  return id;
}

// ---------- 演示 ----------
console.log('=== 收件箱 ===');
console.table(getInbox.all());

console.log('=== 标签统计 ===');
console.table(getTags.all());

console.log('=== 搜索「灵感」 ===');
console.table(searchNotes.all('灵感'));

console.log('=== 新建灵感 ===');
const newId = createNote('SQLite 原型跑通了，可以接前端了。', ['产品', '方法']);
console.log('new note id =', newId);

console.log('=== 新建后收件箱 ===');
console.table(getInbox.all());
```

运行：

```bash
sqlite3 muse.db < muse_schema.sql
sqlite3 muse.db < muse_seed.sql
node demo.js
```

## 6. 与前端原型字段映射

| 前端字段 | 数据库来源 |
|---|---|
| `id` | `muse_notes.id`（雪花 ID，前端用 `BigInt` 或字符串） |
| `content` | `muse_notes.content` |
| `tags` | `v_muse_notes_full.tags_text`，按 `,` 分割 |
| `project` | `v_muse_notes_full.project_name` |
| `status` | `muse_notes.status` |
| `source` | `muse_notes.source` |
| `archived` | `muse_notes.archived_at IS NOT NULL` |
| `createdAt` | `muse_notes.created_at` |
| `updatedAt` | `muse_notes.updated_at` |

### 前端适配建议

```js
// 雪花 ID 转字符串，避免 Number 精度丢失
function normalizeNote(row) {
  return {
    id: String(row.id),                // 转字符串
    content: row.content,
    tags: row.tags_text ? row.tags_text.split(',') : [],
    project: row.project_name || '',
    status: row.status,
    source: row.source,
    archived: !!row.archived_at,
    createdAt: row.created_at,
    updatedAt: row.updated_at,
  };
}
```

## 9. 一键生成脚本

```bash
#!/bin/bash
# build_muse_db.sh

set -e

DB=muse.db

echo "▶ 删除旧数据库…"
rm -f $DB $DB-wal $DB-shm

echo "▶ 建表…"
sqlite3 $DB < muse_schema.sql

echo "▶ 灌入种子数据…"
sqlite3 $DB < muse_seed.sql

echo "▶ 校验…"
sqlite3 $DB "SELECT '表:', COUNT(*) FROM sqlite_master WHERE type='table' AND name LIKE 'muse_%';"
sqlite3 $DB "SELECT '视图:', COUNT(*) FROM sqlite_master WHERE type='view' AND name LIKE 'v_muse_%';"
sqlite3 $DB "SELECT '灵感:', COUNT(*) FROM muse_notes;"
sqlite3 $DB "SELECT '标签:', COUNT(*) FROM muse_tags;"
sqlite3 $DB "SELECT 'FTS 行数:', COUNT(*) FROM muse_notes_fts;"
sqlite3 $DB "PRAGMA integrity_check;"

echo "✅ 完成：$DB"
```

执行：

```bash
chmod +x build_muse_db.sh
./build_muse_db.sh
```
